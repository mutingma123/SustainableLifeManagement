\section{Background}

\begin{frame}{Motivation}
	\begin{itemize}
		\item Cognitive overload and limited attention make time allocation fragile \citep{Robinson2024, Grant2019}.
		\item Sustainable planning needs explicit tradeoffs under finite time \citep{Hart2024}.
		\item Team settings add coordination and capacity coupling across agents.
	\end{itemize}
\end{frame}

\section{Problem Formulation}

\begin{frame}{Problem Setting}
	\begin{itemize}
		\item Tasks are items, agents are knapsacks with capacities $C_j$.
		\item Task to agent assignments face stochastic sizes $S_{ij}$ and returns $R_{ij}$.
		\item Overload is controlled by chance constraints on capacity violations.
	\end{itemize}
\end{frame}

\begin{frame}{Stochastic Multiple Knapsack}
	\begin{align*}
		\max \quad & \sum_{i \in \mathcal{I}} \sum_{j \in \mathcal{J}} \E\left[R_{ij}\right] x_{ij} \\
		\text{s.t.} \quad & \sum_{j \in \mathcal{J}} x_{ij} \leq 1 \quad \forall i \in \mathcal{I} \\
		& \Prob\left(\sum_{i \in \mathcal{I}} S_{ij} x_{ij} \leq C_j\right) \geq 1 - \epsilon_j \quad \forall j \in \mathcal{J}.
	\end{align*}
	\vspace{0.2cm}
	\small{Sizes and returns can be correlated, and risk is limited through chance constraints.}
\end{frame}

\section{Takeaways}

\begin{frame}{Key Takeaways}
	\begin{itemize}
		\item The formulation unifies cognitive resource allocation with stochastic knapsack structure.
		\item Chance constraints control overload risk while retaining assignment flexibility.
		\item Decomposition and data driven methods are a natural fit for scaling the model.
	\end{itemize}
\end{frame}
